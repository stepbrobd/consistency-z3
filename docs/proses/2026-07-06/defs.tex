\newcommand*{\squigglyarrowhead}[0]{\ \raisebox{0.15em}{\tikz \draw [->,
      line join=round,
      line width=0.8pt,
      decorate, decoration={
        zigzag,
        segment length=6,
        amplitude=.9,
        post=lineto,
        post length=1pt
}] (0,0) -- (0.4,0);}\ }
\newcommand*{\squigglyarrowtail}[0]{\ \raisebox{0.15em}{\tikz \draw [-,
      line join=round,
      line width=0.8pt,
      decorate, decoration={
        zigzag,
        segment length=6,
        amplitude=.9,
        post=lineto,
        post length=1pt
}] (0,0) -- (0.4,0);}\ }
